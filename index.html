import cv2 from pyzbar.pyzbar import decode def parse_cccd_qr(raw_bytes): """ Giải mã chuỗi Bytes UTF-8 và tách thông tin CCCD """ try: # Decode UTF-8 chuẩn tiếng Việt text = raw_bytes.decode('utf-8') except UnicodeDecodeError: text = raw_bytes.decode('latin-1', errors='ignore') fields = text.split('|') # Định dạng QR CCCD chip tiêu chuẩn gồm 7 trường if len(fields) >= 7: return { "Trạng thái": "Thành công", "Số CCCD": fields[0], "Số CMND cũ": fields[1] if fields[1] else "Không có", "Họ và tên": fields[2], "Ngày sinh": f"{fields[3][:2]}/{fields[3][2:4]}/{fields[3][4:]}", "Giới tính": fields[4], "Địa chỉ": fields[5], "Ngày cấp": f"{fields[6][:2]}/{fields[6][2:4]}/{fields[6][4:]}" } return {"Trạng thái": "Thất bại", "Lỗi": "Sai định dạng CCCD", "Raw": text} def read_qr_from_image(image_path): img = cv2.imread(image_path) if img is None: print("Lỗi: Không thể mở file ảnh!") return # Giải mã mã QR từ ảnh barcodes = decode(img) if not barcodes: print("Không tìm thấy mã QR nào trong ảnh.") return for barcode in barcodes: # Lấy trực tiếp raw bytes từ pyzbar result = parse_cccd_qr(barcode.data) print("--- THÔNG TIN CĂN CƯỚC CÔNG DÂN ---") for key, value in result.items(): print(f"{key}: {value}") # --- CHẠY THỬ --- if __name__ == "__main__": # Thay 'cccd.jpg' bằng đường dẫn file ảnh của bạn read_qr_from_image("cccd.jpg")
